# UI Parser

> Extracted from discussion_guikit.txt - JSON-based UI definitions and dynamic loading

## Overview

This document covers the UI Parser system for GUIKit, which enables:
- JSON-based UI definitions stored on SD card
- Dynamic UI loading at runtime
- External generation of interfaces (by other programs)
- Clear separation between code and data
- On-demand resource loading to save memory

All implementations follow Objective-C style memory management (NOT ARC/reference counting) for ESP8266 compatibility.

---

## Architecture

### System Folder Structure on SD Card

```
system/
├── ui/                      # User interfaces (JSON)
│   ├── main_ui.json         # Main interface
│   ├── settings_ui.json     # Settings interface
│   └── login_ui.json        # Login interface
│
├── dict/                    # Dictionaries for text correction
│   ├── fr.txt              # French dictionary
│   ├── en.txt              # English dictionary
│   └── custom.txt          # Custom dictionary
│
├── config/                 # System configurations
│   ├── passwords.txt       # Passwords (hashed or encrypted)
│   ├── styles.json         # Custom styles
│   └── settings.json       # General settings
│
└── logs/                   # System logs (optional)
    └── system.log          # System logs
```

### Benefits

1. **Code/Data Separation**: UI definitions are separate from code
2. **External Generation**: UIs can be generated by other programs
3. **Dynamic Loading**: Load UIs on demand, saving RAM
4. **Memory Efficiency**: Only load necessary data into RAM
5. **Flexibility**: Change UI without recompiling code

### Implementation (ui_parser.cpp)

```c
#include "ui_parser.h"
#include "widget.h"
#include "widget_macros.h"
#include <ArduinoJson.h>  // Lightweight JSON library

// ========== GLOBAL VARIABLES ==========
static char ui_buffer[UI_BUFFER_SIZE + 1];

// ========== JSON PARSING ==========

// Parse UI from JSON string
Widget* ui_parse_json(const char* json) {
    if (!json) return NULL;
    
    DynamicJsonDocument doc(UI_BUFFER_SIZE);
    DeserializationError error = deserializeJson(doc, json);
    
    if (error) {
        Serial.print("JSON parsing error: ");
        Serial.println(error.c_str());
        return NULL;
    }
    
    // Parse root widget
    JsonObject root = doc.as<JsonObject>();
    return ui_parse_widget_object(root, NULL);
}

// Parse widget from JSON object
Widget* ui_parse_widget_object(JsonObject json, Widget* parent) {
    if (!json.containsKey(JSON_KEY_TYPE)) {
        Serial.println("Error: Missing 'type' in widget");
        return NULL;
    }
    
    const char* type_str = json[JSON_KEY_TYPE];
    WIDGET_TYPE type = ui_string_to_widget_type(type_str);
    
    Widget* widget = NULL;
    
    switch (type) {
        case WIDGET_TYPE_VIEW:
            widget = (Widget*)ui_parse_view(json, parent);
            break;
        case WIDGET_TYPE_BUTTON:
            widget = (Widget*)ui_parse_button(json, parent);
            break;
        case WIDGET_TYPE_LABEL:
            widget = (Widget*)ui_parse_label(json, parent);
            break;
        case WIDGET_TYPE_SLIDER:
            widget = (Widget*)ui_parse_slider(json, parent);
            break;
        case WIDGET_TYPE_CHECKBOX:
            widget = (Widget*)ui_parse_checkbox(json, parent);
            break;
        case WIDGET_TYPE_TEXTFIELD:
            widget = (Widget*)ui_parse_textfield(json, parent);
            break;
        case WIDGET_TYPE_PROGRESS_BAR:
            widget = (Widget*)ui_parse_progress_bar(json, parent);
            break;
        default:
            widget = new_widget(type);
            break;
    }
    
    if (!widget) return NULL;
    
    // Parse common properties
    ui_parse_common_properties(widget, json);
    
    // Parse children
    if (json.containsKey(JSON_KEY_CHILDREN)) {
        JsonArray children = json[JSON_KEY_CHILDREN];
        for (JsonObject child : children) {
            Widget* child_widget = ui_parse_widget_object(child, widget);
            if (child_widget) {
                widget_add_child(widget, child_widget);
            }
        }
    }
    
    return widget;
}

// Parse common widget properties
void ui_parse_common_properties(Widget* widget, JsonObject json) {
    if (!widget || !json) return;
    
    // Position
    if (json.containsKey(JSON_KEY_X)) {
        widget->rect.position.x = json[JSON_KEY_X];
    }
    if (json.containsKey(JSON_KEY_Y)) {
        widget->rect.position.y = json[JSON_KEY_Y];
    }
    
    // Size
    if (json.containsKey(JSON_KEY_WIDTH)) {
        widget->rect.size.width = json[JSON_KEY_WIDTH];
    }
    if (json.containsKey(JSON_KEY_HEIGHT)) {
        widget->rect.size.height = json[JSON_KEY_HEIGHT];
    }
    
    // Text
    if (json.containsKey(JSON_KEY_TEXT)) {
        const char* text = json[JSON_KEY_TEXT];
        strncpy(widget->text.text, text, MAX_TEXT_LENGTH - 1);
        widget->text.text[MAX_TEXT_LENGTH - 1] = '\0';
    }
    
    // Style
    if (json.containsKey(JSON_KEY_STYLE)) {
        const char* style_str = json[JSON_KEY_STYLE];
        widget->style.draw_style = ui_string_to_draw_style(style_str);
    }
}

// Convert string to WIDGET_TYPE
WIDGET_TYPE ui_string_to_widget_type(const char* str) {
    if (strcmp(str, "view") == 0) return WIDGET_TYPE_VIEW;
    if (strcmp(str, "button") == 0) return WIDGET_TYPE_BUTTON;
    if (strcmp(str, "label") == 0) return WIDGET_TYPE_LABEL;
    if (strcmp(str, "slider") == 0) return WIDGET_TYPE_SLIDER;
    if (strcmp(str, "checkbox") == 0) return WIDGET_TYPE_CHECKBOX;
    if (strcmp(str, "textfield") == 0) return WIDGET_TYPE_TEXTFIELD;
    if (strcmp(str, "progress_bar") == 0) return WIDGET_TYPE_PROGRESS_BAR;
    return WIDGET_TYPE_VIEW;
}

// Convert string to draw style
uint64_t ui_string_to_draw_style(const char* str) {
    uint64_t style = 0;
    
    if (strstr(str, "rounded_border")) style |= WIDGET_DRAW_STYLE_ROUNDED_BORDER;
    if (strstr(str, "solid_border")) style |= WIDGET_DRAW_STYLE_SOLID_BORDER;
    if (strstr(str, "solid_fill")) style |= WIDGET_DRAW_STYLE_SOLID_FILL;
    if (strstr(str, "drop_shadow")) style |= WIDGET_DRAW_STYLE_DROP_SHADOW;
    if (strstr(str, "transparent")) style |= WIDGET_DRAW_STYLE_TRANSPARENT_FILL;
    
    return style;
}

// Parse view widget
Widget* ui_parse_view(JsonObject json, Widget* parent) {
    Widget* view = new_widget(WIDGET_TYPE_VIEW);
    if (!view) return NULL;
    return view;
}

// Parse button widget
WidgetButton* ui_parse_button(JsonObject json, Widget* parent) {
    WidgetButton* button = new_button();
    if (!button) return NULL;
    
    // Button-specific properties
    if (json.containsKey(JSON_KEY_TEXT)) {
        const char* text = json[JSON_KEY_TEXT];
        strncpy(button->base.text.text, text, MAX_TEXT_LENGTH - 1);
        button->base.text.text[MAX_TEXT_LENGTH - 1] = '\0';
    }
    
    // Parse callback (if present)
    if (json.containsKey(JSON_KEY_CALLBACK)) {
        const char* callback_name = json[JSON_KEY_CALLBACK];
        button->on_click = ui_get_callback(callback_name);
    }
    
    return button;
}

// Parse label widget
WidgetLabel* ui_parse_label(JsonObject json, Widget* parent) {
    WidgetLabel* label = new_label(NULL);
    if (!label) return NULL;
    
    // Label-specific properties
    if (json.containsKey("auto_resize")) {
        label->auto_resize = json["auto_resize"];
    }
    
    return label;
}

// Parse slider widget
WidgetSlider* ui_parse_slider(JsonObject json, Widget* parent) {
    float min_val = 0.0f;
    float max_val = 100.0f;
    float value = 50.0f;
    bool vertical = false;
    
    if (json.containsKey(JSON_KEY_MIN)) min_val = json[JSON_KEY_MIN];
    if (json.containsKey(JSON_KEY_MAX)) max_val = json[JSON_KEY_MAX];
    if (json.containsKey(JSON_KEY_VALUE)) value = json[JSON_KEY_VALUE];
    if (json.containsKey("vertical")) vertical = json["vertical"];
    
    WidgetSlider* slider = new_slider(min_val, max_val, value);
    if (!slider) return NULL;
    
    slider->vertical = vertical;
    
    return slider;
}

// Parse checkbox widget
WidgetCheckbox* ui_parse_checkbox(JsonObject json, Widget* parent) {
    bool checked = false;
    if (json.containsKey(JSON_KEY_VALUE)) checked = json[JSON_KEY_VALUE];
    
    WidgetCheckbox* checkbox = new_checkbox(checked);
    if (!checkbox) return NULL;
    
    return checkbox;
}

// Parse text field widget
WidgetTextField* ui_parse_textfield(JsonObject json, Widget* parent) {
    uint16_t buffer_size = 256;
    TEXTFIELD_STYLE style = TEXTFIELD_STYLE_NORMAL;
    
    if (json.containsKey("buffer_size")) buffer_size = json["buffer_size"];
    if (json.containsKey("style")) {
        const char* style_str = json["style"];
        if (strcmp(style_str, "password") == 0) {
            style = TEXTFIELD_STYLE_PASSWORD;
        }
    }
    
    WidgetTextField* textfield = new_textfield(buffer_size, style);
    if (!textfield) return NULL;
    
    if (json.containsKey(JSON_KEY_TEXT)) {
        const char* text = json[JSON_KEY_TEXT];
        textfield_set_text(textfield, text);
    }
    
    return textfield;
}

// Get callback function by name
void (*ui_get_callback(const char* name))(void) {
    if (strcmp(name, "on_save") == 0) return on_save_click;
    if (strcmp(name, "on_load") == 0) return on_load_click;
    if (strcmp(name, "on_exit") == 0) return on_exit_click;
    return NULL;
}

// ========== UI LOADING ==========

// Load UI from SD card
Widget* ui_load_from_sd(const char* filename) {
    if (!filename) return NULL;
    
    char path[64];
    snprintf(path, sizeof(path), "%s/%s", UI_FOLDER, filename);
    
    if (!sd_file_exists(path)) {
        Serial.print("Error: UI file not found: ");
        Serial.println(path);
        return NULL;
    }
    
    if (!sd_read_file(path, ui_buffer, UI_BUFFER_SIZE)) {
        Serial.print("Error: Cannot read UI file: ");
        Serial.println(path);
        return NULL;
    }
    
    return ui_parse_json(ui_buffer);
}

// Load UI from buffer
Widget* ui_load_from_buffer(const char* json_data) {
    if (!json_data) return NULL;
    return ui_parse_json(json_data);
}

// ========== UI SAVING ==========

// Save UI to SD card
bool ui_save_to_sd(Widget* root, const char* filename) {
    if (!root || !filename) return false;
    
    char path[64];
    snprintf(path, sizeof(path), "%s/%s", UI_FOLDER, filename);
    
    char* json = ui_to_json(root);
    if (!json) return false;
    
    bool result = sd_write_file(path, json);
    free(json);
    return result;
}

// Convert widget tree to JSON
char* ui_to_json(Widget* root) {
    if (!root) return NULL;
    
    DynamicJsonDocument doc(UI_BUFFER_SIZE);
    JsonObject json = doc.to<JsonObject>();
    
    ui_widget_to_json(root, json);
    
    char* result = (char*)malloc(UI_BUFFER_SIZE + 1);
    if (!result) return NULL;
    
    serializeJson(doc, result, UI_BUFFER_SIZE);
    return result;
}

// Convert widget to JSON object
void ui_widget_to_json(Widget* widget, JsonObject json) {
    if (!widget || !json) return;
    
    // Type
    const char* type_str = ui_widget_type_to_string(widget->type);
    json[JSON_KEY_TYPE] = type_str;
    
    // Position
    json[JSON_KEY_X] = widget->rect.position.x;
    json[JSON_KEY_Y] = widget->rect.position.y;
    
    // Size
    json[JSON_KEY_WIDTH] = widget->rect.size.width;
    json[JSON_KEY_HEIGHT] = widget->rect.size.height;
    
    // Text
    if (widget->text.text[0] != '\0') {
        json[JSON_KEY_TEXT] = widget->text.text;
    }
    
    // Style
    const char* style_str = ui_draw_style_to_string(widget->style.draw_style);
    if (style_str) {
        json[JSON_KEY_STYLE] = style_str;
    }
    
    // Type-specific properties
    switch (widget->type) {
        case WIDGET_TYPE_BUTTON: {
            WidgetButton* button = (WidgetButton*)widget;
            // No special properties for button
            break;
        }
        case WIDGET_TYPE_SLIDER: {
            WidgetSlider* slider = (WidgetSlider*)widget;
            json[JSON_KEY_MIN] = slider->min_value;
            json[JSON_KEY_MAX] = slider->max_value;
            json[JSON_KEY_VALUE] = slider->current_value;
            json["vertical"] = slider->vertical;
            break;
        }
        case WIDGET_TYPE_CHECKBOX: {
            WidgetCheckbox* checkbox = (WidgetCheckbox*)widget;
            json[JSON_KEY_VALUE] = checkbox->checked;
            break;
        }
        case WIDGET_TYPE_TEXTFIELD: {
            WidgetTextField* textfield = (WidgetTextField*)widget;
            json["buffer_size"] = textfield->buffer_size;
            const char* style_str = (textfield->style == TEXTFIELD_STYLE_PASSWORD) ? "password" : "normal";
            json["style"] = style_str;
            break;
        }
    }
    
    // Children
    if (widget->children_count > 0) {
        JsonArray children = json.createNestedArray(JSON_KEY_CHILDREN);
        for (uint8_t i = 0; i < widget->children_count; i++) {
            JsonObject child_json = children.createNestedObject();
            ui_widget_to_json(widget->children[i], child_json);
        }
    }
}

// Convert WIDGET_TYPE to string
const char* ui_widget_type_to_string(WIDGET_TYPE type) {
    switch (type) {
        case WIDGET_TYPE_VIEW: return "view";
        case WIDGET_TYPE_BUTTON: return "button";
        case WIDGET_TYPE_LABEL: return "label";
        case WIDGET_TYPE_SLIDER: return "slider";
        case WIDGET_TYPE_CHECKBOX: return "checkbox";
        case WIDGET_TYPE_TEXTFIELD: return "textfield";
        case WIDGET_TYPE_PROGRESS_BAR: return "progress_bar";
        default: return "unknown";
    }
}

// Convert draw style to string
const char* ui_draw_style_to_string(uint64_t style) {
    static char style_str[128];
    style_str[0] = '\0';
    
    if (style & WIDGET_DRAW_STYLE_ROUNDED_BORDER) strcat(style_str, "rounded_border ");
    if (style & WIDGET_DRAW_STYLE_SOLID_BORDER) strcat(style_str, "solid_border ");
    if (style & WIDGET_DRAW_STYLE_SOLID_FILL) strcat(style_str, "solid_fill ");
    if (style & WIDGET_DRAW_STYLE_DROP_SHADOW) strcat(style_str, "drop_shadow ");
    if (style & WIDGET_DRAW_STYLE_TRANSPARENT_FILL) strcat(style_str, "transparent ");
    
    // Remove trailing space
    size_t len = strlen(style_str);
    if (len > 0 && style_str[len - 1] == ' ') {
        style_str[len - 1] = '\0';
    }
    
    return style_str;
}
```

### Supported Widget Types

| JSON Type | C Type | Description |
|-----------|--------|-------------|
| `view` | `Widget` | Container widget |
| `button` | `WidgetButton` | Clickable button |
| `label` | `WidgetLabel` | Text display |
| `slider` | `WidgetSlider` | Value slider |
| `checkbox` | `WidgetCheckbox` | Checkbox |
| `textfield` | `WidgetTextField` | Text input field |
| `progress_bar` | `WidgetProgressBar` | Progress bar |

### Supported Properties

#### Common Properties (All Widgets)

- `type`: Widget type (required)
- `x`: X position
- `y`: Y position
- `width`: Width
- `height`: Height
- `text`: Display text
- `style`: Draw style flags (space-separated)
- `children`: Array of child widgets

#### Button-Specific

- `callback`: Function name to call on click

#### Slider-Specific

- `min`: Minimum value
- `max`: Maximum value
- `value`: Current value
- `vertical`: Orientation (true/false)

#### Checkbox-Specific

- `value`: Check state (true/false)

#### TextField-Specific

- `buffer_size`: Text buffer size
- `style`: `normal` or `password`

## Dictionary Loading

### Load Dictionary from SD Card

```c
bool load_dict_file(const char* filename) {
    if (!filename) return false;
    
    char path[64];
    get_system_path("dict", filename, path, sizeof(path));
    
    if (!sd_file_exists(path)) return false;
    
    uint32_t file_size = sd_get_file_size(path);
    if (file_size == 0) return false;
    
    char* buffer = (char*)malloc(file_size + 1);
    if (!buffer) return false;
    
    if (!sd_read_file(path, buffer, file_size + 1)) {
        free(buffer);
        return false;
    }
    
    // Parse buffer and load words into dictionary
    char* word = strtok(buffer, "\n");
    while (word != NULL) {
        add_to_dictionary(word);
        word = strtok(NULL, "\n");
    }
    
    free(buffer);
    return true;
}
```

### Dictionary File Format

Plain text file with one word per line:

```
bonjour

```

## Usage Examples

### Loading and Displaying a UI

```c
void setup() {
    Serial.begin(115200);
    
    // Initialize hardware
    init_renderer();
    init_touch();
    init_sd_card();
    
    // Create system folders if they don't exist
    create_system_folders();
    
    // Load main UI from SD card
    Widget* main_ui = ui_load_from_sd("main_ui.json");
    if (!main_ui) {
        Serial.println("Error: Cannot load main UI");
        // Fallback to default UI
        main_ui = create_default_ui();
    }
    
    // Draw UI
    draw_widget_tree(main_ui);
}

// Fallback UI if file not found
Widget* create_default_ui(void) {
    Widget* root = new_widget(WIDGET_TYPE_VIEW);
    root->rect.size.width = 320;
    root->rect.size.height = 240;
    
    WidgetLabel* label = new_label("No UI file found");
    label->base.rect.position.x = 50;
    label->base.rect.position.y = 100;
    widget_add_child(root, &label->base);
    
    return root;
}
```

### Dynamic UI Switching

```c
// UI states
typedef enum {
    UI_STATE_MAIN,
    UI_STATE_SETTINGS,
    UI_STATE_LOGIN
} UIState;

UIState current_state = UI_STATE_MAIN;
Widget* current_ui = NULL;

// Load UI for current state
void load_current_ui(void) {
    // Free current UI
    if (current_ui) {
        ui_free(current_ui);
        current_ui = NULL;
    }
    
    // Load new UI
    switch (current_state) {
        case UI_STATE_MAIN:
            current_ui = ui_load_from_sd("main_ui.json");
            break;
        case UI_STATE_SETTINGS:
            current_ui = ui_load_from_sd("settings_ui.json");
            break;
        case UI_STATE_LOGIN:
            current_ui = ui_load_from_sd("login_ui.json");
            break;
    }
    
    if (!current_ui) {
        current_ui = create_default_ui();
    }
    
    draw_widget_tree(current_ui);
}

// Switch to settings UI
void show_settings(void) {
    current_state = UI_STATE_SETTINGS;
    load_current_ui();
}

// Switch to login UI
void show_login(void) {
    current_state = UI_STATE_LOGIN;
    load_current_ui();
}

// Return to main UI
void show_main(void) {
    current_state = UI_STATE_MAIN;
    load_current_ui();
}
```

## Memory Management

### Object Pooling for Parsed UIs

```c
// UI pool for parsed interfaces
#define MAX_UI_POOL 5
UIDefinition ui_pool[MAX_UI_POOL];
bool ui_pool_used[MAX_UI_POOL] = {false};

// Load UI from pool
Widget* ui_pool_load(const char* filename) {
    for (int i = 0; i < MAX_UI_POOL; i++) {
        if (!ui_pool_used[i]) {
            ui_pool_used[i] = true;
            strncpy(ui_pool[i].name, filename, 31);
            ui_pool[i].json_data = NULL;
            
            Widget* root = ui_load_from_sd(filename);
            if (root) {
                ui_pool[i].root = root;
                ui_pool[i].loaded = true;
                return root;
            }
            ui_pool_used[i] = false;
            return NULL;
        }
    }
    return NULL;  // Pool exhausted
}

// Free UI from pool
void ui_pool_free(const char* filename) {
    for (int i = 0; i < MAX_UI_POOL; i++) {
        if (ui_pool_used[i] && strcmp(ui_pool[i].name, filename) == 0) {
            if (ui_pool[i].root) {
                free_widget(ui_pool[i].root);
                ui_pool[i].root = NULL;
            }
            if (ui_pool[i].json_data) {
                free(ui_pool[i].json_data);
                ui_pool[i].json_data = NULL;
            }
            ui_pool_used[i] = false;
            return;
        }
    }
}
```

### Static Buffer for UI Loading

```c
// Single buffer for all UI loading operations
char ui_load_buffer[UI_BUFFER_SIZE + 1];
```
